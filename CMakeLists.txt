cmake_minimum_required(VERSION 3.20)
project(ParlaHaiku VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
set(NLOHMANN_JSON_INCLUDE_DIR "${CMAKE_BINARY_DIR}/include" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT EXISTS "${NLOHMANN_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  file(DOWNLOAD https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp
    "${NLOHMANN_JSON_INCLUDE_DIR}/nlohmann/json.hpp"
    EXPECTED_HASH SHA256=9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6 TLS_VERIFY ON)
endif()
add_library(rpc src/RpcTransport.cpp)
target_include_directories(rpc PUBLIC src "${NLOHMANN_JSON_INCLUDE_DIR}")
target_link_libraries(rpc PUBLIC Threads::Threads)
target_compile_options(rpc PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_SYSTEM_NAME STREQUAL "Haiku")
  add_executable(Parla src/Main.cpp src/MainWindow.cpp src/Widgets.cpp)
  target_link_libraries(Parla PRIVATE rpc be tracker translation)
  target_compile_options(Parla PRIVATE -Wall -Wextra)
  find_program(RC_EXECUTABLE rc REQUIRED)
  find_program(XRES_EXECUTABLE xres REQUIRED)
  add_custom_command(TARGET Parla POST_BUILD
    COMMAND "${RC_EXECUTABLE}" -o "${CMAKE_CURRENT_BINARY_DIR}/Parla.rsrc" "${CMAKE_CURRENT_SOURCE_DIR}/data/Parla.rdef"
    COMMAND "${XRES_EXECUTABLE}" -o $<TARGET_FILE:Parla> "${CMAKE_CURRENT_BINARY_DIR}/Parla.rsrc")
  install(TARGETS Parla RUNTIME DESTINATION apps/Parla)
  set(CORE_BINARY "${CMAKE_CURRENT_SOURCE_DIR}/build/core/target/release/deltachat-rpc-server" CACHE FILEPATH "Built Haiku RPC server")
  install(PROGRAMS "${CORE_BINARY}" DESTINATION apps/Parla)
  install(FILES COPYING data/nlohmann-LICENSE DESTINATION documentation/packages/parla)
endif()
if(BUILD_TESTING)
  add_executable(rpc-test tests/rpc_test.cpp)
  target_link_libraries(rpc-test PRIVATE rpc)
  add_executable(fake-rpc tests/fake_rpc.cpp)
  target_link_libraries(fake-rpc PRIVATE rpc)
  add_test(NAME transport COMMAND rpc-test $<TARGET_FILE:fake-rpc>)
  set_tests_properties(transport PROPERTIES TIMEOUT 20)
endif()
